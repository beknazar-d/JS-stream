// Глобальные функции тестов платформы (как в Vitest).
interface Matchers<R = void> {
  toBe(expected: unknown): R;
  toEqual(expected: unknown): R;
  toStrictEqual(expected: unknown): R;
  toBeCloseTo(expected: number, digits?: number): R;
  toBeTruthy(): R;
  toBeFalsy(): R;
  toBeNull(): R;
  toBeUndefined(): R;
  toBeDefined(): R;
  toBeNaN(): R;
  toBeTypeOf(type: 'string' | 'number' | 'boolean' | 'object' | 'function' | 'undefined' | 'bigint' | 'symbol'): R;
  toBeInstanceOf(expected: Function): R;
  toBeGreaterThan(n: number): R;
  toBeGreaterThanOrEqual(n: number): R;
  toBeLessThan(n: number): R;
  toBeLessThanOrEqual(n: number): R;
  toContain(item: unknown): R;
  toContainEqual(item: unknown): R;
  toHaveLength(length: number): R;
  toHaveProperty(path: string | string[], value?: unknown): R;
  toMatch(pattern: RegExp | string): R;
  toMatchObject(expected: object): R;
  toThrow(expected?: string | RegExp | Function): R;
  toHaveBeenCalled(): R;
  toHaveBeenCalledTimes(times: number): R;
  toHaveBeenCalledWith(...args: unknown[]): R;
}
interface Assertion extends Matchers {
  not: Matchers;
  resolves: Matchers<Promise<void>> & { not: Matchers<Promise<void>> };
  rejects: Matchers<Promise<void>> & { not: Matchers<Promise<void>> };
}
/** Проверка значения. Второй аргумент — пояснение, которое попадёт в сообщение об ошибке. */
declare function expect(actual: unknown, note?: string): Assertion;
/** Группа тестов. */
declare function describe(name: string, fn: () => void): void;
/** Один тест. */
declare function it(name: string, fn: () => unknown): void;
/** Один тест (синоним it). */
declare function test(name: string, fn: () => unknown): void;
declare function beforeEach(fn: () => unknown): void;
declare function afterEach(fn: () => unknown): void;
/** Мок-функция: запоминает вызовы. */
declare function fn<T extends (...args: any[]) => any>(impl?: T): T & {
  mock: { calls: unknown[][]; results: unknown[] };
  mockReturnValue(value: unknown): any;
  mockImplementation(impl: (...args: any[]) => any): any;
};
/** Запускает файл и возвращает строки, которые он вывел в консоль. */
declare function runSnippet(file: string): string[];
